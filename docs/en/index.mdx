The **XCSH Terraform Provider** enables declarative infrastructure-as-code (IaC) management for F5 Distributed Cloud (XC) platform capabilities. Manage HTTP/TCP load balancers, origin pools, Web Application Firewall (WAF) policies, Multi-Cloud Networking (MCN) sites, and DNS zones directly from Terraform.

## Quick Start Configuration

```hcl
terraform {
  required_version = ">= 1.14"

  required_providers {
    xcsh = {
      source  = "f5-sales-demo/xcsh"
      version = ">= 0.1.0"
    }
  }
}

provider "xcsh" {
  api_url   = "https://<XC_TENANT>.console.ves.volterra.io"
  api_token = var.xcsh_api_token
}

variable "xcsh_api_token" {
  description = "F5 Distributed Cloud API Token"
  type        = string
  sensitive   = true
}

# Declarative Namespace Creation
resource "xcsh_namespace" "example" {
  name = "example-namespace"
}

# Declarative HTTP Load Balancer
resource "xcsh_http_loadbalancer" "example" {
  name      = "example-load-balancer"
  namespace = xcsh_namespace.example.name
  domains   = ["app.example.com"]
}
```
